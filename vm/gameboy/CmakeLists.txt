cmake_minimum_required(VERSION 3.15)
project(GameBoy LANGUAGES CXX OBJCXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)

set(OPTIMIZE_FLAGS "-O2 -march=native -flto=thin -fno-rtti")

# set(SANITIZER_FLAGS "-fsanitize=address -fsanitize=undefined")
# set(SANITIZER_FLAGS "-fsanitize=thread")
# set(SANITIZER_FLAGS "-fsanitize=undefined")

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${SANITIZER_FLAGS} ${OPTIMIZE_FLAGS}")
set(CMAKE_OBJCXX_FLAGS "${CMAKE_OBJCXX_FLAGS} ${SANITIZER_FLAGS} ${OPTIMIZE_FLAGS}")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} ${SANITIZER_FLAGS} ${OPTIMIZE_FLAGS}")

set(RAYLIB_DIR "/usr/local/Cellar/raylib" CACHE PATH "Path to raylib")


find_package(raylib REQUIRED HINTS ${RAYLIB_DIR})
find_package(SDL2 REQUIRED COMPONENTS SDL2)


include_directories(include bridge)


set(SOURCES
        src/Cheats.cpp
        src/APU.cpp
        src/Joypad.cpp
        src/Resumable.cpp
        src/CPU.cpp
        src/PPU.cpp
        src/Memory.cpp
        src/Timer.cpp
        main.cpp
        metal_api/metal_api.mm
)

set(RESOURCES metal_api/Shaders.metallib)

add_executable(${PROJECT_NAME} MACOSX_BUNDLE ${SOURCES} ${RESOURCES})

set_source_files_properties(${RESOURCES} PROPERTIES
        MACOSX_PACKAGE_LOCATION Resources
)

target_link_libraries(${PROJECT_NAME} PRIVATE
        raylib
        SDL2::SDL2
        "$<LINK_LIBRARY:FRAMEWORK,IOKit>"
        "$<LINK_LIBRARY:FRAMEWORK,CoreFoundation>"
        "$<LINK_LIBRARY:FRAMEWORK,Foundation>"

        "-framework Cocoa"
        "-framework Metal"
        "-framework QuartzCore"
        "-framework CoreVideo"
        objc
)
